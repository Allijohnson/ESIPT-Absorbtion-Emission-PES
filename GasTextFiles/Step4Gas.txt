#!/bin/bash
#SBATCH --partition=mu3c
#SBATCH --ntasks=1
#SBATCH --cpus-per-task=14
#SBATCH --mem=16G
#SBATCH --job-name=04-MN-es-opt-conform

set -e

sif_file=${SIF_FILES}/g16_centos-7-6.sif
local_scratch=/scratch/$USER/$SLURM_JOBID
mkdir -p $local_scratch
storage_path=${SLURM_SUBMIT_DIR:-$(pwd)}

export BASEFILE=$SLURM_JOB_NAME.$SLURM_JOBID
storage_path=$SLURM_SUBMIT_DIR

echo "BASEFILE: $BASEFILE"
echo "storage_path: $storage_path"

cleanup() {
    echo "Job caught exit signal. Cleaning up split temporary rwf files on $(hostname)..."
    rm -rf $local_scratch
}
trap cleanup EXIT INT TERM ERR

if [ ! -f $storage_path/02-MN-conform.chk ]; then
  echo "ERROR: 02-MN-conform.chk not found - did step 2 complete successfully?"
  exit 1
fi

cd $storage_path


cat <<@EOF >$BASEFILE.sh
#!/bin/bash
export GAUSS_SCRDIR=$local_scratch
cd $storage_path
echo "Running excited state Opt+Freq..."
g16 <$BASEFILE.in >$BASEFILE.out 2>$BASEFILE.err
echo "g16 exit code: \$?"
@EOF

chmod +x $BASEFILE.sh

cat <<"@EOF" >$BASEFILE.in
%oldchk=02-MN-conform.chk
%chk=04-MN-conform.chk
%nprocshared=14
%mem=12GB
# CAM-B3LYP/6-31+G(d,p) TD=(NStates=6,Root=1) Geom=Check Guess=Read Opt=CalcFC Freq NoSymm 

MN conform: excited state opt + freq
Symmetry broken in excited state, using NoSymm and CalcFC

0 1

@EOF

echo "Input file written, starting Singularity..."
echo "Watch progress with: tail -f $storage_path/$BASEFILE.out"
singularity run -B $SLURM_SUBMIT_DIR,$local_scratch $sif_file ./$BASEFILE.sh

echo "Job complete. Running extraction..."

# Python segment to process Gaussian output and extract Emission Energies and Frequencies

python3 - <<'EOF'
import os

basefile = os.environ.get('BASEFILE')
logfile = basefile + ".out"
summary_file = "MN-conform-gas-energy-summary.txt"

emission_ev = None
wavelength_nm = None
frequencies = []

if os.path.exists(logfile):
    with open(logfile, 'r') as f:
        for line in f:
            if "Excited State   1:" in line:
                try:
                    parts = line.split()
                    emission_ev = float(parts[4])
                    wavelength_nm = float(parts[6])
                except (IndexError, ValueError):
                    pass
            elif "Frequencies --" in line:
                line_freqs = line.split()[2:]
                frequencies.extend(line_freqs)
    with open(summary_file, 'a') as out:
        out.write("\n=========================================\n")
        out.write("Step 3: Normal Emission Energy and Freq Check (S1)\n")
        out.write("Job Name: " + str(basefile) + "\n")
        out.write("-----------------------------------------\n")
        
        if emission_ev is not None:
            out.write("Emission Energy (eV):            " + "{:.4f}".format(emission_ev) + " eV\n")
            out.write("Emission Wavelength (nm):        " + "{:.2f}".format(wavelength_nm) + " nm\n")
        else:
            out.write("Emission Data:                   Not found (check convergence).\n")
            
        out.write("\n--- Excited-State Vibrational Frequencies (cm^-1) ---\n")
        if frequencies:
            out.write(", ".join(frequencies) + "\n")
            
            # Check for imaginary frequencies in S1 state
            imag_freqs = [f for f in frequencies if float(f) < 0]
            if imag_freqs:
                out.write("\nImaginary Frequency " + str(len(imag_freqs)) + " NOT a minimum\n")
            else:
                out.write("\nNo imaginary frequencies! This is a minimum\n")
        else:
            out.write("No excited-state frequency calculations found.\n")
            
    print("Step 4 data appended to " + summary_file + " successfully")
else:
    print("Error: Output file " + logfile + " not found.")
EOF
