#!/bin/bash
#SBATCH --partition=mu3c
#SBATCH --nodes=1
#SBATCH --ntasks=1
#SBATCH --cpus-per-task=16
#SBATCH --mem=16G
#SBATCH --job-name=02-MN-td-conform

set -e

sif_file=${SIF_FILES}/g16_centos-7-6.sif
local_scratch=/scratch/$USER/$SLURM_JOBID
mkdir -p $local_scratch
storage_path=${SLURM_SUBMIT_DIR:-$(pwd)}

export BASEFILE=$SLURM_JOB_NAME.$SLURM_JOBID
storage_path=$SLURM_SUBMIT_DIR

echo "BASEFILE: $BASEFILE"
echo "storage_path: $storage_path"

cleanup() {
    echo "Job caught exit signal. Cleaning up split temporary rwf files on \$(hostname)..."
    rm -rf $local_scratch
}
trap cleanup EXIT INT TERM ERR

if [ ! -f $storage_path/01-MN-conform.chk ]; then
  echo "ERROR: 01-MN-conform.chk not found - did step 1 complete successfully?"
  exit 1
fi

cd $storage_path

cat <<@EOF >$BASEFILE.sh
#!/bin/bash
export GAUSS_SCRDIR=$local_scratch
cd $storage_path
echo "Running TD-DFT vertical excitation (linear response)..."
g16 <$BASEFILE.in >$BASEFILE.out 2>$BASEFILE.err
echo "g16 exit code: \$?"
@EOF

chmod +x $BASEFILE.sh

cat <<"@EOF" >$BASEFILE.in
%oldchk=01-MN-conform.chk
%chk=02-MN-conform.chk
%nprocshared=16
%mem=14GB
# CAM-B3LYP/6-31+G(d,p) TD=(NStates=6) SCRF=(Solvent=isolve) Geom=Check Guess=Read

MN conform absorption: linear response vertical excited states

0 1

@EOF

echo "Input file written, starting Singularity..."
echo "Watch progress with: tail -f $storage_path/$BASEFILE.out"
singularity run -B $SLURM_SUBMIT_DIR,$local_scratch $sif_file ./$BASEFILE.sh

echo "Job complete. Running extraction..."

# Python segment to process Gaussian output and extract SCF energies and Frequencies

python3 - <<'EOF'
import os, re

basefile = os.environ.get('BASEFILE', '02-MN-td-conform')
logfile = basefile + ".out"
summary_file = "MN-isolve-energy-summary.txt"

energy_re  = re.compile(r"Total Energy, E\(TD-HF/TD-DFT\)\s*=\s*([-\d.]+)")
s1_re      = re.compile(r"Excited State\s+1:.*?([\d.]+) eV\s+([\d.]+) nm.*?f=([\d.]+)")
opt_head   = re.compile(r"Optimized Parameters")
coord_re   = re.compile(r"!\s+\w+\s+R\(14,27\)\s+([\d.]+)")   
coord_re2  = re.compile(r"!\s+\w+\s+R\(27,14\)\s+([\d.]+)")

excitation_ev = None
wavelength_nm = None
excited_states = {}

if os.path.exists(logfile):
    with open(logfile, 'r') as s:
        for line in s:
            if "Excited State" in line and "Singlet" in line:
                try:
                    parts = line.split()
                    state_num = int(parts[2].rstrip(':'))
                    ev = float(parts[4])
                    nm = float(parts[6])
                    f_val = float(line.split('f=')[1].split()[0])
                    if state_num <= 4:
                        excited_states[state_num] = (ev, nm, f_val)
                except (IndexError, ValueError):
                    pass
    if 1 in excited_states:
        excitation_ev = excited_states[1][0] if excited_states else None
        wavelength_nm = excited_states[1][1] if excited_states else None

    with open(summary_file, 'a') as out:
        if excitation_ev is not None:
            out.write("\n~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~\n")
            out.write("Step 2: Vertical Excitation Energies of MN\n")
            out.write("Excitation Energy (eV):          " + "{:.4f}".format(excitation_ev) + " eV\n")
            out.write("Excitation Wavelength (nm):      " + "{:.2f}".format(wavelength_nm) + " nm\n")
        if excited_states:
            for state_num in sorted(excited_states.keys()):
                ev, nm, f_val = excited_states[state_num]
                out.write("Excited State {}:  {:.4f} eV  {:.2f} nm  f={:.4f}\n".format(
                    state_num, ev, nm, f_val))
            print("Success! Excitation data saved to " + summary_file)
        else:
            out.write("\n~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~\n")
            out.write("No 'Excited State   1:' data found. Check your TD keyword.\n")
            print("Warning: No excitation data found.")
else:
    print("Error: Output file " + logfile + " not found.")
EOF
