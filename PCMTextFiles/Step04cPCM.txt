#!/bin/bash
#SBATCH --partition=mu3c
#SBATCH --ntasks=1
#SBATCH --cpus-per-task=16
#SBATCH --mem=16G
#SBATCH --job-name=04c-MN-S1PT-opt

set -e

sif_file=${SIF_FILES}/g16_centos-7-6.sif
local_scratch=/scratch/$USER/$SLURM_JOBID
mkdir -p $local_scratch
storage_path=${SLURM_SUBMIT_DIR:-$(pwd)}

export BASEFILE=$SLURM_JOB_NAME.$SLURM_JOBID
storage_path=$SLURM_SUBMIT_DIR

echo "BASEFILE: $BASEFILE"
echo "storage_path: $storage_path"

cleanup() {
    echo "Job caught exit signal. Cleaning up split temporary rwf files on \$(hostname)..."
    rm -rf $local_scratch
}
trap cleanup EXIT INT TERM ERR

if [ ! -f $storage_path/04b-MN-ESIPT-scan.chk ]; then
  echo "ERROR: 04b-MN-ESIPT-scan.chk not found — did step 04 (S1-NF opt) complete successfully?"
  exit 1
fi

cd $storage_path


cat <<@EOF >$BASEFILE.sh
#!/bin/bash
export GAUSS_SCRDIR=$local_scratch
cd $storage_path
echo "Running in isolve S1-PT tautomer optimization + freq..."
g16 <$BASEFILE.in >$BASEFILE.out 2>$BASEFILE.err
echo "g16 exit code: \$?"
@EOF

chmod +x $BASEFILE.sh

cat <<@EOF >$BASEFILE.in
%oldchk=04b-MN-ESIPT-scan.chk
%chk=04c-MN-S1PT-opt.chk
%nprocshared=16
%mem=14GB
# CAM-B3LYP/6-31+G(d,p) TD=(NStates=5,Root=1) SCRF=(Solvent=isolve)
  Opt=(CalcFC,MaxCycles=100) NoSymm

MN in isolve: S1 tautomer (PT form) geometry optimization
Starting from lowest enegy of ESIPT scan (04b)

0 1
insert geometry here

--Link1--
%chk=04c-MN-S1PT-opt.chk
%nprocshared=16
%mem=16GB
# CAM-B3LYP/6-31+G(d,p) TD=(NStates=5,Root=1) SCRF=(Solvent=isolve)
  Geom=Check Guess=Read Freq NoSymm

MN in isolve: S1 tautomer (PT form) analytical frequency check
Running on the fully optimized S1-Keto geometry from step 1

0 1

@EOF

echo "Input file written, starting Singularity..."
echo "Watch progress with: tail -f $storage_path/$BASEFILE.out"
singularity run -B $SLURM_SUBMIT_DIR,$local_scratch $sif_file ./$BASEFILE.sh

echo "Job complete. Running extraction..."

# Python segment to confirm minimum and extract S1-PT energy
python3 - <<'EOF'
import os

basefile     = os.environ.get('BASEFILE', '04c-MN-S1PT-opt')
logfile      = basefile + ".out"
summary_file = "MN-isolve-energy-summary.txt"

opt_converged   = False
emission_ev     = None
wavelength_nm   = None
total_energy    = None
frequencies     = []
imag_freqs = []


if os.path.exists(logfile):
    with open(logfile, 'r') as f:
        for line in f:
            if "Optimization completed" in line:
                opt_converged = True

            if "Total Energy, E(TD-HF/TD-DFT)" in line:
                try:
                    total_energy = float(line.split("=")[1].strip().split()[0])
                except (IndexError, ValueError):
                    pass

            if "Excited State   1:" in line:
                try:
                    parts         = line.split()
                    emission_ev   = float(parts[4])
                    wavelength_nm = float(parts[6])
                except (IndexError, ValueError):
                    pass

            if "Frequencies --" in line:
                line_freqs = line.split()[2:]
                frequencies.extend(line_freqs)

    with open(summary_file, 'a') as out:
        out.write("\n~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~\n")
        out.write("Step 04c: in isolve S1-PT Tautomer Opt + Freq\n")
        out.write("Job Name: " + str(basefile) + "\n")
        out.write("~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~\n")

        out.write("Optimization converged: " + ("YES" if opt_converged else "NO — check geometry") + "\n")

        if total_energy is not None:
            out.write("S1-PT Total Energy (a.u.):       {:.6f}\n".format(total_energy))
        else:
            out.write("S1-PT Total Energy:              Not found.\n")

        if emission_ev is not None:
            out.write("S1 Energy (eV):                  {:.4f} eV\n".format(emission_ev))
            out.write("S1 Wavelength (nm):              {:.2f} nm\n".format(wavelength_nm))
        else:
            out.write("S1 Energy:                       Not found.\n")

        out.write("\n--- Vibrational Frequencies (cm^-1) ---\n")
        if frequencies:
            out.write(", ".join(frequencies) + "\n")
            for f in frequencies:
                try:
                    if float(f) <0:
                        imag_freqs.append(f)
                except ValueError:
                    pass
            if imag_freqs:
                out.write("\n Imaginary Frequencies Found " + str(len(imag_freqs)) + " NOT a minimum\n")
                out.write("\n Frequencies: " + ", ".join(imag_freqs) + "\n")
            else:
                out.write("\nYay! No imaginary frequencies, this is a minimum\n")
        else:
            out.write("No frequencies found.\n")
    print("Step 04c data appended to " + summary_file + " successfully.")
else:
    print("ERROR: Output file " + logfile + " not found.")
EOF
